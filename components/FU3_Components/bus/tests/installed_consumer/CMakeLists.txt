cmake_minimum_required(VERSION 3.16)
project(fu3_installed_consumer LANGUAGES CXX)
find_package(fu3-bus CONFIG REQUIRED)
add_executable(fu3_consumer main.cpp)
target_link_libraries(fu3_consumer PRIVATE fu3::bus_system)
enable_testing()
foreach(path RANGE 0 3)
  add_test(NAME installed_consumer_${path} COMMAND fu3_consumer ${path})
  set_tests_properties(installed_consumer_${path} PROPERTIES TIMEOUT 10)
endforeach()
