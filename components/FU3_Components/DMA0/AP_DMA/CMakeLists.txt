cmake_minimum_required(VERSION 3.16)
project(fu3_ap_dma LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SYSTEMC_HOME "" CACHE PATH "SystemC 2.3.4 installation prefix")
find_path(SYSTEMC_INCLUDE_DIR systemc HINTS "${SYSTEMC_HOME}/include" "$ENV{SYSTEMC_HOME}/include")
find_library(SYSTEMC_LIBRARY systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64" "${SYSTEMC_HOME}/lib-linux64" "$ENV{SYSTEMC_HOME}/lib" "$ENV{SYSTEMC_HOME}/lib64" "$ENV{SYSTEMC_HOME}/lib-linux64")
if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
  message(FATAL_ERROR "SystemC 2.3.4 not found. Set SYSTEMC_HOME or SYSTEMC_INCLUDE_DIR and SYSTEMC_LIBRARY.")
endif()
file(STRINGS "${SYSTEMC_INCLUDE_DIR}/sysc/kernel/sc_ver.h" _systemc_version REGEX "^#define SYSTEMC_VERSION[ \t]+20221128")
if(NOT _systemc_version)
  message(FATAL_ERROR "The selected headers are not SystemC 2.3.4 (20221128).")
endif()
find_package(Threads REQUIRED)
add_library(fu3_ap_dma STATIC src/ap_dma.cpp)
add_library(cdc::components::fu3_ap_dma ALIAS fu3_ap_dma)
target_include_directories(fu3_ap_dma PUBLIC include "${SYSTEMC_INCLUDE_DIR}")
target_link_libraries(fu3_ap_dma PUBLIC "${SYSTEMC_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(fu3_ap_dma PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  add_executable(test_fu3_ap_dma_registers tests/test_registers.cpp)
  target_link_libraries(test_fu3_ap_dma_registers PRIVATE fu3_ap_dma)
  add_test(NAME fu3_ap_dma_registers COMMAND test_fu3_ap_dma_registers)
  set_tests_properties(fu3_ap_dma_registers PROPERTIES TIMEOUT 10)
endif()
