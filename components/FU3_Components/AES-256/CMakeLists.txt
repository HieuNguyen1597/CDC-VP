cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fu3_aes LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET SystemC::systemc)
  set(SYSTEMC_HOME "/opt/systemc-2.3.4" CACHE PATH "SystemC installation prefix")
  find_path(SYSTEMC_INCLUDE_DIR systemc HINTS "${SYSTEMC_HOME}/include")
  find_library(SYSTEMC_LIBRARY systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64")
  if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
    message(FATAL_ERROR "Set SYSTEMC_HOME to an installed SystemC prefix")
  endif()
  find_package(Threads REQUIRED)
  add_library(SystemC::systemc UNKNOWN IMPORTED)
  set_target_properties(SystemC::systemc PROPERTIES
    IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}"
    INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS}")
endif()

add_library(fu3_aes src/aes_core.cpp src/aes_controller.cpp)
add_library(fu3::aes ALIAS fu3_aes)
target_compile_features(fu3_aes PUBLIC cxx_std_17)
target_include_directories(fu3_aes PUBLIC include)
target_link_libraries(fu3_aes PUBLIC SystemC::systemc)
option(FU3_AES_COVERAGE "Instrument AES library sources with GCC gcov" OFF)
if(FU3_AES_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "FU3_AES_COVERAGE requires GCC/gcov")
  endif()
  target_compile_options(fu3_aes PRIVATE --coverage -O0 -g)
  target_link_options(fu3_aes PUBLIC --coverage)
endif()
if(MSVC)
  target_compile_options(fu3_aes PRIVATE /W4)
else()
  target_compile_options(fu3_aes PRIVATE -Wall -Wextra -Wpedantic)
endif()

set(_aes_tests ${BUILD_TESTING})
if(DEFINED CDC_BUILD_TESTS)
  set(_aes_tests ${CDC_BUILD_TESTS})
endif()
option(FU3_AES_BUILD_TESTS "Build AES VP regression" ${_aes_tests})
if(FU3_AES_BUILD_TESTS)
  enable_testing()
  add_executable(test_fu3_aes_spec tests/test_hds_core.cpp)
  target_link_libraries(test_fu3_aes_spec PRIVATE fu3::aes)
  foreach(scenario encrypt128 decrypt128 encrypt256 decrypt256 upper_key128)
    add_test(NAME fu3_aes_spec_${scenario} COMMAND test_fu3_aes_spec ${scenario})
    set_tests_properties(fu3_aes_spec_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;aes;spec")
  endforeach()
  add_executable(test_fu3_aes_trace tests/test_fips_trace.cpp)
  target_link_libraries(test_fu3_aes_trace PRIVATE fu3::aes)
  foreach(bits 128 256)
    add_test(NAME fu3_aes_spec_trace${bits} COMMAND test_fu3_aes_trace ${bits}
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/fips197_rounds.txt")
    set_tests_properties(fu3_aes_spec_trace${bits} PROPERTIES TIMEOUT 30 LABELS "fu3;aes;spec")
  endforeach()
  add_executable(test_fu3_aes tests/test_aes.cpp)
  target_link_libraries(test_fu3_aes PRIVATE fu3::aes)
  foreach(scenario registers aes128 aes256 protocol commands reset timing result_mux
      strobes status_flow access_matrix concurrency reset_edges reset_entry latency_1 latency_2 latency_17 latency_257
      busy_matrix reset_boundaries debug_invariants hds_fields)
    add_test(NAME fu3_aes_policy_${scenario} COMMAND test_fu3_aes ${scenario})
    set_tests_properties(fu3_aes_policy_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;aes;vp_policy")
  endforeach()
  add_executable(test_fu3_aes_core tests/test_core.cpp)
  target_link_libraries(test_fu3_aes_core PRIVATE fu3::aes)
  foreach(scenario core init_zero block_zero)
    add_test(NAME fu3_aes_policy_${scenario} COMMAND test_fu3_aes_core ${scenario})
    set_tests_properties(fu3_aes_policy_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;aes;vp_policy;corners")
  endforeach()
  option(FU3_AES_ORACLE_TESTS "Generate independent OpenSSL reference vectors" OFF)
  if(FU3_AES_ORACLE_TESTS)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(FU3_AES_OPENSSL_EXECUTABLE openssl)
    if(NOT FU3_AES_OPENSSL_EXECUTABLE)
      message(FATAL_ERROR "FU3_AES_ORACLE_TESTS requires the openssl executable")
    endif()
    set(_aes_vectors "${CMAKE_CURRENT_BINARY_DIR}/openssl-vectors.txt")
    add_custom_command(OUTPUT "${_aes_vectors}"
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_vectors.py"
        --openssl "${FU3_AES_OPENSSL_EXECUTABLE}" --output "${_aes_vectors}"
      DEPENDS scripts/generate_vectors.py VERBATIM)
    add_custom_target(fu3_aes_vectors DEPENDS "${_aes_vectors}")
    add_dependencies(test_fu3_aes_core fu3_aes_vectors)
    add_dependencies(test_fu3_aes fu3_aes_vectors)
    add_test(NAME fu3_aes_reference_core_oracle COMMAND test_fu3_aes_core "${_aes_vectors}")
    foreach(bits 128 256)
      add_test(NAME fu3_aes_reference_mmio_oracle_${bits} COMMAND test_fu3_aes oracle${bits} "${_aes_vectors}")
    endforeach()
    add_test(NAME fu3_aes_reference_sequence_oracle COMMAND test_fu3_aes sequence_oracle "${_aes_vectors}")
    set_tests_properties(fu3_aes_reference_core_oracle fu3_aes_reference_mmio_oracle_128 fu3_aes_reference_mmio_oracle_256
      fu3_aes_reference_sequence_oracle
      PROPERTIES TIMEOUT 60 LABELS "fu3;aes;reference")
  endif()
  option(FU3_AES_BUS_TESTS "Build real FU3 bus integration test" ON)
  if(FU3_AES_BUS_TESTS)
    if(NOT TARGET fu3_bus_system)
      set(FU3_BUS_BUILD_TESTS OFF)
      add_subdirectory(../bus "${CMAKE_CURRENT_BINARY_DIR}/fu3-bus")
    endif()
    add_executable(test_fu3_aes_bus tests/test_aes.cpp)
    target_compile_definitions(test_fu3_aes_bus PRIVATE FU3_AES_BUS_TEST=1)
    target_link_libraries(test_fu3_aes_bus PRIVATE fu3::aes fu3::bus_system)
    add_test(NAME fu3_aes_policy_bus COMMAND test_fu3_aes_bus aes256)
    add_test(NAME fu3_aes_policy_bus_aes128 COMMAND test_fu3_aes_bus aes128)
    add_test(NAME fu3_aes_policy_bus_lanes COMMAND test_fu3_aes_bus bus_lanes)
    set_tests_properties(fu3_aes_policy_bus fu3_aes_policy_bus_aes128 fu3_aes_policy_bus_lanes
      PROPERTIES TIMEOUT 30 LABELS "fu3;aes;vp_policy;integration")
  endif()
endif()
