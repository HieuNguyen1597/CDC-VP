cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fu3_gpio_io LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET SystemC::systemc)
  set(SYSTEMC_HOME "/opt/systemc-2.3.4" CACHE PATH "SystemC installation")
  find_path(SYSTEMC_INCLUDE_DIR systemc HINTS "${SYSTEMC_HOME}/include")
  find_library(SYSTEMC_LIBRARY systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64")
  if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
    message(FATAL_ERROR "Set SYSTEMC_HOME to an installed SystemC prefix")
  endif()
  find_package(Threads REQUIRED)
  add_library(SystemC::systemc UNKNOWN IMPORTED)
  set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}")
  set_property(TARGET SystemC::systemc PROPERTY INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS}")
  message(STATUS "FU3 GPIO SystemC headers: ${SYSTEMC_INCLUDE_DIR}")
  message(STATUS "FU3 GPIO SystemC library: ${SYSTEMC_LIBRARY}")
endif()
add_library(fu3_gpio_io src/gpio.cpp src/io_pad.cpp src/subsystem.cpp)
add_library(fu3::gpio_io ALIAS fu3_gpio_io)
target_compile_features(fu3_gpio_io PUBLIC cxx_std_17)
target_include_directories(fu3_gpio_io PUBLIC include)
target_link_libraries(fu3_gpio_io PUBLIC SystemC::systemc)
if(MSVC)
  target_compile_options(fu3_gpio_io PRIVATE /W4)
else()
  target_compile_options(fu3_gpio_io PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(FU3_GPIO_COVERAGE "Enable GCC coverage" OFF)
if(FU3_GPIO_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "FU3_GPIO_COVERAGE requires GCC/gcov")
  endif()
  target_compile_options(fu3_gpio_io PRIVATE --coverage -O0 -g)
  target_link_options(fu3_gpio_io PUBLIC --coverage)
endif()
set(_fu3_gpio_tests_default ${BUILD_TESTING})
if(DEFINED CDC_BUILD_TESTS)
  set(_fu3_gpio_tests_default ${CDC_BUILD_TESTS})
endif()
option(FU3_GPIO_BUILD_TESTS "Build GPIO/IOSEL tests" ${_fu3_gpio_tests_default})
if(FU3_GPIO_BUILD_TESTS)
  enable_testing()
  add_executable(test_fu3_gpio_io tests/test_gpio_io.cpp)
  target_link_libraries(test_fu3_gpio_io PRIVATE fu3::gpio_io)
  foreach(scenario registers sampling interrupts wake protocol reset concurrency reset_pulse iosel integration random)
    add_test(NAME fu3_gpio_${scenario} COMMAND test_fu3_gpio_io ${scenario})
    set_tests_properties(fu3_gpio_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;gpio;io")
  endforeach()
  add_executable(test_fu3_gpio_spec tests/test_spec.cpp)
  target_link_libraries(test_fu3_gpio_spec PRIVATE fu3::gpio_io)
  if(MSVC)
    target_compile_options(test_fu3_gpio_spec PRIVATE /W4)
  else()
    target_compile_options(test_fu3_gpio_spec PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  foreach(scenario register_outputs input_stream irq_boundaries wake_priority clock_reset pad_truth pad_random address_decode)
    add_test(NAME fu3_gpio_spec_${scenario} COMMAND test_fu3_gpio_spec ${scenario})
    set_tests_properties(fu3_gpio_spec_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;gpio;io;spec")
  endforeach()
endif()
